    Through Leads.

    Its definition
    VAR DirectSpend =
    CALCULATE (
        SUM ( fact_ad_spend_allocated_daily[spend] ),
        fact_ad_spend_allocated_daily[allocation_tier] = "direct_campus"
    )
VAR ProvinceOnlyAllocation =
    SUMX (
        VALUES ( dim_channel[channel_key] ),
        VAR ChanProvSpend =
            CALCULATE (
                SUM ( fact_ad_spend_allocated_daily[spend] ),
                fact_ad_spend_allocated_daily[allocation_tier] = "province_only",
                ALL ( dim_franchise ),
                ALL ( dim_campus )
            )
        VAR ChanProvLeads =
            CALCULATE ( [Leads], ALL ( dim_franchise ), ALL ( dim_campus ) )
        VAR ChanLocLeads = [Leads]
        RETURN ChanProvSpend * DIVIDE ( ChanLocLeads, ChanProvLeads )
    )
VAR UnattributedAllocation =
    SUMX (
        VALUES ( dim_channel[channel_key] ),
        VAR ChanUnattribSpend =
            CALCULATE (
                SUM ( fact_ad_spend_allocated_daily[spend] ),
                fact_ad_spend_allocated_daily[allocation_tier] = "unattributed",
                ALL ( dim_franchise ),
                ALL ( dim_campus ),
                ALL ( dim_province )
            )
        VAR ChanAllLeads =
            CALCULATE (
                [Leads],
                ALL ( dim_franchise ),
                ALL ( dim_campus ),
                ALL ( dim_province )
            )
        VAR ChanLocLeads = [Leads]
        RETURN ChanUnattribSpend * DIVIDE ( ChanLocLeads, ChanAllLeads )
    )
RETURN DirectSpend + ProvinceOnlyAllocation + UnattributedAllocation
